cmake_minimum_required(VERSION 3.24)
project(REAzipNative VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(REAZIP_BUILD_GUI "Build Qt Quick application" ON)
set(REAZIP_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/.deps" CACHE PATH "Bootstrapped dependencies")
set(REAZIP_ENGINE_DIR "" CACHE PATH "Verified native engine directory")
if(NOT EXISTS "${REAZIP_DEPS}/bit7z/CMakeLists.txt")
  message(FATAL_ERROR "Run bootstrap/dependencies.py first")
endif()
set(BIT7Z_CUSTOM_7ZIP_PATH "${REAZIP_DEPS}/7zip" CACHE STRING "" FORCE)
set(BIT7Z_AUTO_FORMAT ON CACHE BOOL "" FORCE)
set(BIT7Z_USE_NATIVE_STRING ON CACHE BOOL "" FORCE)
set(BIT7Z_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BIT7Z_USE_PCH OFF CACHE BOOL "" FORCE)
add_subdirectory("${REAZIP_DEPS}/bit7z" "${CMAKE_CURRENT_BINARY_DIR}/bit7z")
add_executable(reazip-probe tests/foundation/probe.cpp)
target_link_libraries(reazip-probe PRIVATE bit7z)
if(MSVC)
  target_compile_options(reazip-probe PRIVATE /utf-8)
endif()
set_target_properties(reazip-probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(WIN32)
  set(REAZIP_ENGINE_FILE "7z.dll")
else()
  set(REAZIP_ENGINE_FILE "7z.so")
endif()
if(NOT EXISTS "${REAZIP_ENGINE_DIR}/${REAZIP_ENGINE_FILE}")
  message(FATAL_ERROR "REAZIP_ENGINE_DIR must contain the matching complete native engine")
endif()
add_test(NAME native-foundation COMMAND reazip-probe "${REAZIP_ENGINE_DIR}/${REAZIP_ENGINE_FILE}" "${CMAKE_BINARY_DIR}/test-data")
set_tests_properties(native-foundation PROPERTIES TIMEOUT 60)
if(REAZIP_BUILD_GUI)
  find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Test)
  qt_standard_project_setup(REQUIRES 6.8)
  qt_add_executable(reazip app/main.cpp)
  qt_add_qml_module(reazip URI REAzip VERSION 1.0 QML_FILES ui/Main.qml)
  target_link_libraries(reazip PRIVATE Qt6::Quick Qt6::QuickControls2)
  set_target_properties(reazip PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" MACOSX_BUNDLE TRUE MACOSX_BUNDLE_GUI_IDENTIFIER "com.reazip.app")
endif()
# Module expansion points are explicit and owned by their task.
foreach(module engine/read engine/write worker jobs platform settings themes i18n models)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${module}/CMakeLists.txt")
    add_subdirectory("${module}")
  endif()
endforeach()
