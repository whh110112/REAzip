cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION.txt" ZENZIP_VERSION_FULL)
string(STRIP "${ZENZIP_VERSION_FULL}" ZENZIP_VERSION_FULL)
string(REGEX REPLACE "-.*" "" ZENZIP_VERSION_NUMBER "${ZENZIP_VERSION_FULL}")
project(ZENzipNative VERSION ${ZENZIP_VERSION_NUMBER} LANGUAGES CXX)
if(WIN32)
  enable_language(RC)
  configure_file(platform/windows.rc.in "${CMAKE_BINARY_DIR}/windows.rc" @ONLY)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(ZENZIP_BUILD_GUI "Build Qt Quick application" ON)
set(ZENZIP_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/.deps" CACHE PATH "Bootstrapped dependencies")
set(ZENZIP_ENGINE_DIR "" CACHE PATH "Verified native engine directory")
if(NOT EXISTS "${ZENZIP_DEPS}/bit7z/CMakeLists.txt")
  message(FATAL_ERROR "Run bootstrap/dependencies.py first")
endif()
set(BIT7Z_CUSTOM_7ZIP_PATH "${ZENZIP_DEPS}/7zip" CACHE STRING "" FORCE)
set(BIT7Z_AUTO_FORMAT ON CACHE BOOL "" FORCE)
set(BIT7Z_USE_NATIVE_STRING ON CACHE BOOL "" FORCE)
set(BIT7Z_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BIT7Z_USE_PCH OFF CACHE BOOL "" FORCE)
if(WIN32)
  set(BIT7Z_AUTO_PREFIX_LONG_PATHS ON CACHE BOOL "" FORCE)
endif()
add_subdirectory("${ZENZIP_DEPS}/bit7z" "${CMAKE_CURRENT_BINARY_DIR}/bit7z")
set(CMAKE_CXX_STANDARD 17)
add_executable(zenzip-probe tests/foundation/probe.cpp)
target_compile_features(zenzip-probe PRIVATE cxx_std_17)
target_link_libraries(zenzip-probe PRIVATE bit7z)
if(MSVC)
  target_compile_options(zenzip-probe PRIVATE /utf-8)
endif()
set_target_properties(zenzip-probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(WIN32)
  set(ZENZIP_ENGINE_FILE "7z.dll")
else()
  set(ZENZIP_ENGINE_FILE "7z.so")
endif()
if(NOT EXISTS "${ZENZIP_ENGINE_DIR}/${ZENZIP_ENGINE_FILE}")
  message(FATAL_ERROR "ZENZIP_ENGINE_DIR must contain the matching complete native engine")
endif()
add_test(NAME native-foundation COMMAND zenzip-probe "${ZENZIP_ENGINE_DIR}/${ZENZIP_ENGINE_FILE}" "${CMAKE_BINARY_DIR}/test-data" "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/sample.iso")
set_tests_properties(native-foundation PROPERTIES TIMEOUT 60)
if(ZENZIP_BUILD_GUI)
  find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Test)
  qt_standard_project_setup(REQUIRES 6.8)
  qt_add_executable(zenzip app/main.cpp)
  target_compile_features(zenzip PRIVATE cxx_std_17)
  target_compile_definitions(zenzip PRIVATE ZENZIP_VERSION_FULL="${ZENZIP_VERSION_FULL}")
  qt_add_qml_module(zenzip URI ZENzip VERSION 1.0 QML_FILES ui/Main.qml ui/Pill.qml ui/Sheet.qml)
  target_link_libraries(zenzip PRIVATE Qt6::Quick Qt6::QuickControls2)
  set_target_properties(zenzip PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" MACOSX_BUNDLE TRUE MACOSX_BUNDLE_GUI_IDENTIFIER "com.zenzip.app")
  if(WIN32)
    set_target_properties(zenzip PROPERTIES WIN32_EXECUTABLE TRUE)
  endif()
  if(APPLE)
    set_target_properties(zenzip PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/platform/Info.plist.in" MACOSX_BUNDLE_SHORT_VERSION_STRING "${ZENZIP_VERSION_NUMBER}" MACOSX_BUNDLE_BUNDLE_VERSION "${ZENZIP_VERSION_NUMBER}")
  endif()
endif()
# Module expansion points are explicit and owned by their task.
foreach(module engine/read engine/write worker jobs platform settings themes i18n vault models)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${module}/CMakeLists.txt")
    add_subdirectory("${module}")
  endif()
endforeach()
if(WIN32)
  foreach(target zenzip zenzip-worker zenzip-probe zenzip-read-test zenzip-write-test zenzip-journal-test zenzip-security-test zenzip-disk-test zenzip-large-test zenzip-jobs-test zenzip-platform-test zenzip-settings-test zenzip-vault-test zenzip-integration-test zenzip-model-test zenzip-gui-test)
    if(TARGET ${target})
      target_sources(${target} PRIVATE "${CMAKE_BINARY_DIR}/windows.rc")
      if(MSVC)
        target_link_options(${target} PRIVATE /MANIFEST:NO)
      endif()
    endif()
  endforeach()
endif()
if(TARGET zenzip)
  add_custom_command(TARGET zenzip POST_BUILD COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:zenzip>/engine" COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ZENZIP_ENGINE_DIR}/${ZENZIP_ENGINE_FILE}" "$<TARGET_FILE_DIR:zenzip>/engine/${ZENZIP_ENGINE_FILE}")
  if(APPLE)
    add_dependencies(zenzip zenzip-worker)
    add_custom_command(TARGET zenzip POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:zenzip-worker>" "$<TARGET_FILE_DIR:zenzip>/zenzip-worker")
  endif()
  include(packaging/Install.cmake)
endif()
